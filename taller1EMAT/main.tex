    \item $f_i$: frecuencia absoluta de $x_i$
    \item $n$: tamaño de la muestra
\end{itemize}

\subsection*{4. Cuartiles}
\begin{equation*}
    Q_j = L_i + a \cdot \frac{\left(\dfrac{j \cdot m}{4}\right) - F_{i-1}}{f_i}
\end{equation*}

\textbf{Notación:}
\begin{itemize}
    \item $L_i$: valor $x_i$ de la fila donde se ubica la posición (al ser datos no agrupados, $L_i = x_i$)
    \item $a$: amplitud de clase; para datos no agrupados $a = 0$
    \item $m$: tamaño de la muestra ($m=22$)
    \item $j$: número del cuartil a calcular ($j=1,2,3$)
    \item $F_{i-1}$: frecuencia acumulada de la fila anterior
    \item $f_i$: frecuencia absoluta de la fila donde se ubica la posición
\end{itemize}

\section*{Medidas de Dispersión}

A partir de la distribución de frecuencias de los estudiantes de EMAT B, organizamos los cálculos para datos con frecuencia ($f_i$):

\begin{enumerate}

    \item \textbf{Cálculo de la Media Aritmética ($\bar{x}$):}

        \begin{table}[H]

    $$\bar{x} = \frac{\sum f_i x_i}{n} = \frac{317}{22} \approx 14.41 \text{ horas}$$

    \item \textbf{Cálculo de la Varianza ($S^2$):}
    Utilizando la fórmula práctica para datos clasificados:

    \begin{table}[H]
    \centering
    \caption{Cantidad de horas programando de los estudiantes de EMAT B (horas enteras)}
    \label{tab:fi-xi2-horas}
    \begin{tabular}{@{}cccc@{}}

        15 & 3 & 225 & 675 \\

        \textbf{Total} & \textbf{22} & -- & $\mathbf{7527}$ \\
        \bottomrule
    \end{tabular}
\end{table}


    $$S^2 = \frac{1}{n} \sum f_i x_i^2 - (\bar{x})^2$$
    Sustituyendo los valores obtenidos ($\sum f_i x_i^2 = 7527$ y $n = 22$):
    $$S^2 = \frac{7527}{22} - (14.41)^2$$
    $$S^2 = 342.14 - 207.65 = 134.49 \text{ horas}^2$$

    \item \textbf{Cálculo de la Desviación Estándar ($S$):}
    $$S = \sqrt{S^2} = \sqrt{134.49} \approx 11.60 \text{ horas}$$

    \item \textbf{Cálculo del Coeficiente de Variación ($CV$):}
    $$CV = \frac{S}{\bar{x}} = \frac{11.60}{14.41} \approx 0.8049 \implies 80.49\%$$
    \textit{Interpretación:} Al ser un coeficiente de variación mayor al $30\%$, indica que existe una alta dispersión y heterogeneidad en la cantidad de horas que los estudiantes dedican a programar.
\end{enumerate}

\section*{Medidas de distribución y de forma}

Evaluamos la asimetría y el apuntamiento para los datos de los estudiantes de EMAT B (con $\bar{x} = 14.41$, Mediana $Me = 10$ al ubicarse en el acumulado, Desviación Estándar $S = 11.60$, cuartiles $Q_1 = 5$, $Q_3 = 22$ y percentiles aproximados $P_{10} = 2$, $P_{90} = 30$):

\begin{enumerate}
    \item \textbf{Coeficiente de Asimetría (Pearson - $AS_2$):}
    $$AS_2 = \frac{3(\bar{x} - Me)}{S} = \frac{3(14.41 - 10)}{11.60} = \frac{13.23}{11.60} \approx 1.14$$
    \textbf{Interpretación:} Como $AS_2 > 0$, la distribución presenta una \textbf{asimetría positiva} (sesgada hacia la derecha, con mayor concentración de estudiantes en pocas horas de programación y algunos valores altos que elevan la media).

    \item \textbf{Coeficiente de Curtosis ($K$):}
    Calculando $Q = \frac{Q_3 - Q_1}{2} = \frac{22 - 5}{2} = 8.5$:
    $$K = \frac{Q}{P_{90} - P_{10}} = \frac{8.5}{30 - 2} = \frac{8.5}{28} \approx 0.30$$
    \textbf{Interpretación:} Dado que $K \approx 0.30$ se ubica cerca de $0.25$, la curva de distribución tiende a ser \textbf{mesocúrtica}.
\end{enumerate}

\section*{Probabilidad}
 
\subsection*{Contexto general del experimento}
 
\noindent
\begin{tabularx}{\textwidth}{@{}l X@{}}
\textbf{Experimento:} & Elegir un plumón al azar de un estuche de 12 plumones de colores. \\[6pt]
\textbf{Espacio muestral ($S$):} & $S = $ \{azul, amarillo, rojo, verde, negro, blanco, anaranjado, morado, rosado, marrón, celeste, gris\}, con $n(S) = 12$. \\[6pt]
\textbf{Evento ($A$):} & Escoger un color primario del estuche de plumones de colores: $A = $ \{amarillo, azul, rojo\}. \\[6pt]
\end{tabularx}
 
\subsection*{Evento Simple}
 
\noindent
\begin{tabularx}{\textwidth}{@{}l X@{}}
\textbf{Experimento:} & Extraer un único plumón del estuche, en un solo intento. \\[6pt]
\textbf{Espacio muestral ($S$):} & El mismo espacio muestral general, $n(S) = 12$. \\[6pt]
\textbf{Evento ($B$):} & Obtener un plumón de color azul: $B = \{\text{azul}\}$, con $n(B) = 1$. \\[6pt]
\end{tabularx}
 
\noindent Al haber solo 1 plumón azul de un total de 12:
$$P(B) = \frac{n(B)}{n(S)} = \frac{1}{12} \approx 0.0833 \quad (8.33\%)$$
 
\noindent\textbf{Interpretación:} Al elegir un plumón al azar del estuche de colores, la probabilidad de que este sea de color azul es del $8.33\%$.
 
\subsection*{Evento Compuesto}
 
\noindent
\begin{tabularx}{\textwidth}{@{}l X@{}}
\textbf{Experimento:} & Extraer dos plumones de manera consecutiva y \textit{sin reposición}: primero un plumón amarillo, y a continuación un plumón rojo. \\[6pt]
\textbf{Espacio muestral:} & Para el primer intento, $S_1$ es el espacio muestral general ($n(S_1) = 12$). Para el segundo intento, dado que no hay reposición, $S_2$ contiene los 11 plumones restantes ($n(S_2) = 11$). \\[6pt]
\textbf{Evento ($D$):} & Obtener un plumón amarillo en el primer intento: $D = \{\text{amarillo}\}$. \\[6pt]
\textbf{Evento ($E$):} & Obtener un plumón rojo en el segundo intento, dado que el primero fue amarillo: $E = \{\text{rojo} \mid \text{amarillo}\}$. \\[6pt]
\textbf{Evento compuesto ($C$):} & $C = D \cap E$. \\[6pt]
\end{tabularx}
 
\noindent Al depender el segundo resultado del primero, aplicamos la definición de \textbf{probabilidad condicional} y su regla de multiplicación para eventos dependientes:
$$P(D \cap E) = P(D) \cdot P(E \mid D)$$
 
\begin{itemize}
    \item Probabilidad de sacar amarillo en el primer intento: $P(D) = \dfrac{1}{12}$
    \item Probabilidad condicional de sacar rojo en el segundo intento, dado que el primero fue amarillo: $P(E \mid D) = \dfrac{1}{11}$
\end{itemize}
 
\noindent Multiplicando ambas probabilidades según la fórmula:
$$P(C) = P(D \cap E) = \frac{1}{12} \times \frac{1}{11} = \frac{1}{132} \approx 0.00757 \quad (0.76\%)$$
 
\noindent\textbf{Interpretación:} La probabilidad de extraer primero un plumón amarillo y, de manera consecutiva y sin reposición, un plumón rojo es del $0.76\%$.
